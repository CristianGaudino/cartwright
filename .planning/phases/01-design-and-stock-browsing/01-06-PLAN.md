```typescript
// Toggle one value of a multi-select group; resets page to 1; removing a make also drops models whose facet parent is that make.
export function toggle(q: StockQuery, group: "make" | "model" | "body" | "fuel", value: string, facets: Facets | null): StockQuery;
// One entry per applied filter, in UI-SPEC group order; `next` is the query with that one filter removed (page 1).
export function appliedFilters(q: StockQuery, facets: Facets | null): { id: string; label: string; next: StockQuery }[];
// The rows a multi-select group renders. Options in the given order, then any selected value missing from them (label = raw value).
// counted true: count shown, and disabled when count is 0 and the value is not selected. counted false: no count, never disabled.
export function facetRows(options: FacetOption[], selected: string[], counted: boolean): { value: string; label: string; count?: number; disabled: boolean }[];
```

This plan adds to `useStock()` in `src/components/stock/StockShell.tsx`: `facets: Facets | null`, the facets of the last result that loaded in this session (equal to `summary.facets` whenever `summary` is not null; null only when no load has succeeded yet).
