// src/lib/leads/schema.ts  (client-safe: zod only, no server imports)
export const CONSENT: { version: string; text: string };       // text = "I agree to Carright using these details to reply to my request."
export const LIMITS: { name: 100; phone: 30; email: 200; message: 1000 };
export const leadSchema;                                         // z.discriminatedUnion("type", [...]); this plan: the "enquiry" branch
export type LeadInput = z.infer<typeof leadSchema>;
export type LeadType = LeadInput["type"];                        // grows as branches are added
export const TYPE_LABELS: Record<LeadType, string>;              // email subject words, e.g. enquiry -> "Enquiry"

// src/lib/leads/process.ts  (no Next imports)
export type Outcome = "stored" | "notify_failed" | "duplicate" | "invalid" | "captcha" | "rate" | "honeypot" | "error";
export type LeadResult =
  | { status: "ok" }
  | { status: "invalid"; fieldErrors: Record<string, string[]> }
  | { status: "failed"; kind: "captcha" | "rate" | "server" };
export interface LeadRow {
  id: string; clientRef: string; type: LeadType;
  carId: string | null; carSlug: string | null; carTitle: string | null; carPriceEur: number | null;
  sourcePath: string; name: string; phone: string | null; email: string | null; message: string | null;
  details: Record<string, string | number>; photoKeys: string[]; consentAt: Date; consentVersion: string;
}
export type CarSnapshot = Pick<LeadRow, "carId" | "carSlug" | "carTitle" | "carPriceEur">;
export function snapshotCar(carId: string | undefined, getCar: Deps["getCar"]): Promise<CarSnapshot | null>; // no carId -> all four null, getCar not called; unknown or sold -> null (refuse)
export interface LeadStore { exists(clientRef: string): Promise<boolean>; insert(row: LeadRow): Promise<boolean>; markNotified(id: string): Promise<void> } // insert -> false when clientRef already stored (processLead then answers ok, sends no mail)
export interface Mailer { send(m: { subject: string; text: string; replyTo?: string }): Promise<void> }                   // to and from are bound in deps
export interface Deps { store: LeadStore; mailer: Mailer; getCar(id: string): Promise<Car | null>; demo: boolean }       // 03-08 adds limiter and verifyTurnstile
export function processLead(fd: FormData, ctx: { ip: string }, deps: Deps): Promise<LeadResult>;

// src/lib/leads/email.ts
export function buildLeadEmail(row: LeadRow, opts: { demo: boolean }): { subject: string; text: string };

// src/lib/log.ts  (outside the lead lint scope added in 03-08)
export function logLead(e: { leadId?: string; type: LeadType | "n/a"; outcome: Outcome }): void; // console.error for error and notify_failed, console.log otherwise; no other key (D-08)

// src/lib/leads/deps.ts  (import "server-only")
export function getDeps(): Deps;  // fakes only when process.env.E2E_FAKE_BACKENDS === "1"

// src/app/actions/lead.ts  ("use server", async exports only)
export async function submitLead(fd: FormData): Promise<LeadResult>;

// src/components/leads/LeadDialog.tsx  ("use client")
export type CarSummary = { id: string; name: string; title: string; price: string; photo?: string }; // never the Car object
export function LeadActions(props: { car: CarSummary }): JSX.Element;   // the action-area buttons plus the dialog
export function LeadDialog(props: { title: string; open: boolean; onClose(): void; children: React.ReactNode }): JSX.Element;

// src/components/leads/LeadForm.tsx  ("use client")
export type FieldBinding = { id: string; name: string; value: string; onChange(v: string): void; error?: string };
export function LeadForm(props: {
  type: LeadType; car?: CarSummary; variant: "dialog" | "page";
  submitLabel: string; success: { heading: string; body: string };
  children: (form: { bind(name: string): FieldBinding }) => React.ReactNode; // the caller lays out its fields in UI-SPEC order
}): JSX.Element;
