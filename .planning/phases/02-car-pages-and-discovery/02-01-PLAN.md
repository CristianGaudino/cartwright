```typescript
export const SOLD_VISIBLE_DAYS = 30; // one setting (D-02): confirm with the dealer
export const isSold = (car: Pick<Car, "soldAt">) => car.soldAt !== undefined;

export interface Car {
  // existing keys unchanged, then, all optional, in this order:
  engineLitres?: number;   // 2 for "2L"; render "2.0 L"
  colour?: string;
  doors?: number;
  seats?: number;
  owners?: number;         // 2 for "2 Owners"
  nctExpiry?: string;      // "2027-07" for "07/2027"
  maxSpeedMph?: number;    // 126 for "126 mph"
  zeroTo62Secs?: number;   // 9.9 for "9.9 seconds"
  description?: string;    // dealer text, line breaks kept, contact block removed
  features?: string[];     // equipment items
  photos: Photo[];
  soldAt?: string;         // "YYYY-MM-DD"; drives the sold window and 410 only, never rendered
}

export interface StockQuery { /* existing */ sold?: "include" | "only" } // absent = in stock only

export type Lookup = { state: "live"; car: Car } | { state: "gone"; make: string; makeSlug: string } | null;
export interface Taxonomy {
  makes: { slug: string; label: string }[];
  bodies: { slug: string; label: string }[];
  models: { makeSlug: string; slug: string; label: string }[];
}
export interface StockProvider {
  query(q: StockQuery): Promise<StockResult>;
  getBySlug(slug: string): Promise<Car | null>; // in stock or sold inside the period
  getById(id: string): Promise<Car | null>;     // same visibility as getBySlug
  lookup(slug: string): Promise<Lookup>;        // live | gone (sold, past the period) | null (never known)
  taxonomy(): Promise<Taxonomy>;                // over every car, expired included
}

// src/lib/stock/file-provider.ts
export function createFileProvider(input: Car[], opts?: { now?: () => Date }): StockProvider;
```

`src/lib/stock/index.ts` spreads the file provider into `stock`, so `lookup` and `taxonomy` reach pages with no change there.
